    22. 해시 테이블: 해시 함수, 충돌 해결 기법(체이닝, 개방 주소법)¶

    직접 AI 없이 직접 할 수 있어야 하는 항목

    배열은 번호를 알면 바로 찾고, 연결 리스트는 처음부터 훑어야 찾는다(20번(자료구조의 필요성)). 이름이나 단어처럼 번호가 아닌 키로도 바로 찾고 싶을 때 쓰는 것이 해시 테이블이다. 키를 계산해 칸 번호로 바꾸는 것이 핵심이다. 사전, 집합, 캐시, 데이터베이스 색인 곳곳에 쓰인다. 직접형 항목이므로 해시 테이블을 AI 없이 짜고, 넣기와 찾기를 칸 번호로 손으로 따라갈 수 있어야 한다. 예제는 C다.

    이 항목에서 배우는 것¶

    • 해시 함수가 키를 칸 번호로 바꾸는 과정을 계산할 수 있다.
    • 충돌이 났을 때 체이닝과 선형 탐사(개방 주소법)가 각각 어떻게 넣고 찾는지 손으로 따라갈 수 있다.
    • 적재율이 높아질 때 느려지는 이유를 설명하고, 지우기와 크기 늘리기의 함정을 피할 수 있다.

    1. 해시 함수가 키를 칸 번호로 바꾼다¶

    unsigned int hash(const char *s) {
    unsigned int h = 0;
    while (*s != '\0')
        h = h * 31 + (unsigned char)*s++;   /* 글자 코드를 섞어 넣는다 */
    return h;
}

    printf("%u\n", hash("kim"));        /* 106191 */
printf("%u\n", hash("kim") % 7);    /* 1 */
    • 해시 함수는 키를 정수(해시값)로 바꾸고, 칸 수 m으로 나눈 나머지가 칸 번호가 된다. 같은 키는 언제나 같은 칸으로 가므로, 찾을 때 그 칸만 보면 된다. 그래서 넣기와 찾기가 평균 O(1)이다.
    • 좋은 해시 함수는 비슷한 키도 칸을 고르게 나눠 쓰게 한다. unsigned int는 넘쳐도 2³²로 나눈 나머지만 남는 것이 C의 약속이라, 긴 키도 문제없다.

    2. 충돌은 피할 수 없다: 체이닝과 선형 탐사¶

    직접 해 보기해시 테이블 실험실키 단추를 눌러 해시 테이블에 넣고 찾아보며, 해시값이 칸 번호가 되는 계산과 충돌이 났을 때 체이닝과 선형 탐사가 각각 어떻게 처리하는지 본다
    새 창에서 크게 열기
    • 칸보다 키의 종류가 훨씬 많으니 서로 다른 키가 같은 칸으로 오는 충돌은 반드시 생긴다. 위젯에서 kim, choi, jung은 m = 7일 때 모두 1번 칸이다.
    • 체이닝은 칸마다 연결 리스트를 두고 같은 칸에 온 키를 줄줄이 단다. 찾을 때는 그 칸의 목록만 훑는다.
    • 선형 탐사(개방 주소법의 하나)는 칸이 차 있으면 다음 칸, 그다음 칸으로 가서 빈칸에 넣는다. 찾을 때도 같은 길을 따라가다가 빈칸을 만나면 없는 것으로 본다. 연결 리스트가 없어 메모리가 붙어 있지만, 표가 다 차면 더 넣을 수 없다.

    3. 칸이 차면 느려진다¶

    • 들어 있는 키 수를 칸 수로 나눈 값을 적재율(α)이라 한다. 선형 탐사는 적재율이 높을수록 빈칸이 드물어져, 한 번 찾을 때 보는 칸이 늘어난다. 없는 키는 빈칸을 만나야 끝나므로 훨씬 크게 늘어난다.
    • 적재율 0.5에서 없는 키는 평균 2.5칸, 0.99에서는 8,328칸을 봤다. 시간도 9~10ns에서 3,280~3,332ns로 늘었다. "평균 O(1)"은 적재율을 낮게 지킬 때의 이야기다.
    • 그래서 해시 테이블은 적재율이 정한 값(흔히 0.5~0.75)을 넘으면 칸을 두 배쯤 늘린다. 칸 번호가 h % m이라 m이 바뀌면 칸 번호도 바뀌므로, 모든 키를 새 표에 다시 넣는다.

    예측해 보기¶

    문제 1. 정수 키의 칸 번호를 k % 7로 정하는 선형 탐사 표에 10, 17, 24, 5를 차례로 넣는다. 무엇이 출력될까?

    #include <stdio.h>

#define M 7
int table[M];                  /* 0은 빈칸 */

void insert(int k) {
    int j = k % M;
    while (table[j] != 0)
        j = (j + 1) % M;
    table[j] = k;
}

int main(void) {
    int keys[] = { 10, 17, 24, 5 };
    for (int i = 0; i < 4; i++)
        insert(keys[i]);
    for (int j = 0; j < M; j++)
        printf("%d ", table[j]);
    printf("\n");
    return 0;
}
    정답과 해설

    0 0 0 10 17 24 5. 10, 17, 24는 모두 7로 나눈 나머지가 3이다. 10은 3번, 17은 3번이 차 있어 4번, 24는 3·4번이 차 있어 5번에 들어간다. 5는 원래 5번 칸인데 24가 차지하고 있어 6번으로 밀린다. 충돌은 같은 나머지끼리만 생기는 것이 아니다. 앞의 충돌로 밀려난 키가 다른 키의 자리를 차지한다.

    문제 2. 문제 1의 표에서 17을 "칸을 비우는" 방법으로 지운 뒤 24를 찾는다. 무엇이 출력될까?

    #include <stdio.h>

#define M 7
int table[M];                  /* 0은 빈칸 */

void insert(int k) {
    int j = k % M;
    while (table[j] != 0)
        j = (j + 1) % M;
    table[j] = k;
}

int find(int k) {
    int j = k % M;
    while (table[j] != 0) {
        if (table[j] == k)
            return j;
        j = (j + 1) % M;
    }
    return -1;                 /* 빈칸을 만나면 없다 */
}

int main(void) {
    int keys[] = { 10, 17, 24, 5 };
    for (int i = 0; i < 4; i++)
        insert(keys[i]);
    table[find(17)] = 0;       /* 17을 지운다: 칸을 비운다 */
    printf("%d\n", find(24));
    return 0;
}
    정답과 해설

    -1. 24는 5번 칸에 그대로 있는데 찾지 못한다. 24를 찾을 때는 3번(10)에서 시작해 4번으로 가는데, 17을 지우며 비운 4번이 빈칸이라 "여기까지 없으면 없다"고 멈춘다. 선형 탐사에서는 칸을 비우지 말고 "지운 자리" 표시를 남겨, 찾을 때는 건너가고 넣을 때는 다시 쓰게 한다(아래 실습 2).

    직접 해보기¶

    실습: AI 없이 해시 테이블 쓰기

    1. 체이닝 해시 테이블로 단어 수를 센다. 칸 7개, 칸마다 { 단어, 개수, next } 노드의 연결 리스트를 둔다. "the cat and the hat and the bat"의 단어를 차례로 넣고, 단어별 개수를 돌려주는 get을 짠다.
    2. 문제 2의 표에 "지운 자리" 표시(DELETED, 여기서는 -1)를 더해 remove를 고친다. 17을 지운 뒤에도 24를 찾아야 한다.
    3. 실습 1의 표에 단어가 칸 수의 0.75배를 넘으면 칸을 두 배 넘게 늘리는 과정을 말로 적는다. 왜 노드를 새 표의 같은 번호 칸으로 옮기면 안 되는지 쓴다.
    정답

    1. 예시. 같은 단어가 있으면 개수만 늘리고, 없으면 그 칸 목록의 맨 앞에 새 노드를 단다. strcpy는 SDL 검사에서 오류가 되므로 snprintf로 복사한다.

      #include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define M 7

struct entry {
    char key[16];
    int count;
    struct entry *next;
};

struct entry *table[M];

unsigned int hash(const char *s) {
    unsigned int h = 0;
    while (*s != '\0')
        h = h * 31 + (unsigned char)*s++;
    return h;
}

void free_table(void) {               /* 칸마다 목록을 따라가며 노드를 돌려준다 */
    for (int i = 0; i < M; i++) {
        struct entry *e = table[i];
        while (e != NULL) {
            struct entry *next = e->next;  /* free하기 전에 다음 노드를 읽어 둔다 */
            free(e);
            e = next;
        }
        table[i] = NULL;
    }
}

void add(const char *word) {
    unsigned int i = hash(word) % M;
    for (struct entry *e = table[i]; e != NULL; e = e->next)
        if (strcmp(e->key, word) == 0) {
            e->count++;
            return;
        }
    struct entry *e = malloc(sizeof(struct entry));
    if (e == NULL)
        return;
    snprintf(e->key, sizeof e->key, "%s", word);
    e->count = 1;
    e->next = table[i];            /* 그 칸 목록의 맨 앞에 단다 */
    table[i] = e;
}

int get(const char *word) {
    for (struct entry *e = table[hash(word) % M]; e != NULL; e = e->next)
        if (strcmp(e->key, word) == 0)
            return e->count;
    return 0;
}

int main(void) {
    const char *words[] = { "the", "cat", "and", "the", "hat", "and", "the", "bat" };
    for (int i = 0; i < 8; i++)
        add(words[i]);
    printf("%d\n", get("the"));     /* 3 */
    printf("%d\n", get("and"));     /* 2 */
    printf("%d\n", get("cat"));     /* 1 */
    printf("%d\n", get("dog"));     /* 0 */
    free_table();
    return 0;
}

    2. 예시. 찾을 때 DELETED는 빈칸이 아니므로 건너간다. 넣을 때는 DELETED 칸을 다시 쓴다.

      #include <stdio.h>

#define M 7
#define DELETED -1
int table[M];                      /* 0은 빈칸, -1은 지운 자리 */

void insert(int k) {
    int j = k % M;
    while (table[j] != 0 && table[j] != DELETED)
        j = (j + 1) % M;
    table[j] = k;
}

int find(int k) {
    int j = k % M;
    while (table[j] != 0) {           /* 지운 자리는 건너간다 */
        if (table[j] == k)
            return j;
        j = (j + 1) % M;
    }
    return -1;
}

void remove_key(int k) {
    int j = find(k);
    if (j >= 0)
        table[j] = DELETED;
}

int main(void) {
    int keys[] = { 10, 17, 24, 5 };
    for (int i = 0; i < 4; i++)
        insert(keys[i]);
    remove_key(17);
    printf("%d\n", find(24));         /* 5 */
    printf("%d\n", find(17));         /* -1 */
    return 0;
}

    3. 칸 번호가 hash % m이라 m이 7에서 17로 바뀌면 같은 키도 다른 칸으로 가야 한다. 옛 칸 번호 그대로 옮기면 찾을 때 새 칸 번호로 가서 못 찾는다. 새 표를 받아 모든 노드를 하나씩 hash % 새 m 칸에 다시 달고, 옛 표를 돌려준다. 칸 수를 소수로 두면 키 값에 규칙이 있을 때도 칸이 덜 몰린다.

    AI와 함께 확인하기¶

    AI 활용 과제: 개방 주소법 해시 테이블 판정하기

    1. AI에게 "C로 정수 키를 쓰는 선형 탐사 해시 테이블을 짜 달라. 넣기, 찾기, 지우기, 적재율이 0.75를 넘으면 크기 늘리기가 있어야 한다"고 한다.
    2. 실행하기 전에 다음 경우의 결과를 먼저 적는다. 키 -3 넣기, 문제 2처럼 지운 뒤 찾기, 크기를 늘린 뒤 전에 넣은 키 찾기.
    3. 실행해 대조하고, 틀린 곳은 실행 결과를 보여 주고 고치게 한다.
    확인할 점

    • 칸 번호를 k % m으로 구하면 음수 키에서 음수 칸이 나온다. C의 %는 부호를 그대로 둔다. 키를 unsigned로 바꾸거나 ((k % m) + m) % m으로 구한다(21번(선형 자료구조)).

      printf("%d\n", -3 % 7);            /* -3 */
printf("%d\n", ((-3 % 7) + 7) % 7);   /* 4 */

    • 지우기에서 칸을 그냥 비우면 문제 2처럼 뒤에 있던 키를 못 찾는다. "지운 자리" 표시를 쓰는지 본다.

    • 크기를 늘릴 때 옛 배열을 새 배열 앞쪽에 그대로 복사하는 코드는 늘린 뒤의 찾기에서 드러난다. 모든 키를 새 크기로 다시 넣어야 한다.
    • "지운 자리"가 많이 쌓이면 빈칸이 줄어 찾기가 느려진다. 다시 넣을 때 지운 자리를 함께 정리하는지도 본다.

    더 알아보기¶

    이 항목의 측정 프로그램

    3절의 값은 칸 2²⁰(1,048,576)개짜리 선형 탐사 표에 서로 다른 무작위 정수 키를 적재율만큼 넣고, 넣은 키 전부와 없는 키 100만 개를 찾으며 탐사 수와 시간을 잰 것이다. 이 PC(Core i9-14900KF, DDR5 64GB)에서 cl /O2로 빌드해 3번 실행했다. 탐사 수는 세 번 모두 같았고, 시간은 그 범위를 적었다. 해시는 키에 2654435761을 곱해 위쪽 20비트를 쓰는 곱셈 해시다.

    #include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <windows.h>

#define BITS 20
#define M (1u << BITS)
#define EMPTY 0u

static unsigned int table[M];
static unsigned int seed = 2026;
static unsigned int rnd(void) { seed ^= seed << 13; seed ^= seed >> 17; seed ^= seed << 5; return seed; }
static unsigned int h(unsigned int k) { return (k * 2654435761u) >> (32 - BITS); }

static double now_ms(void) {
    static LARGE_INTEGER f; LARGE_INTEGER c;
    if (!f.QuadPart) QueryPerformanceFrequency(&f);
    QueryPerformanceCounter(&c);
    return (double)c.QuadPart * 1000.0 / (double)f.QuadPart;
}

int main(void) {
    double loads[] = { 0.5, 0.75, 0.9, 0.95, 0.99 };
    unsigned int *keys = malloc(sizeof(unsigned int) * M);
    unsigned int *miss = malloc(sizeof(unsigned int) * 1000000);
    for (int t = 0; t < 5; t++) {
        memset(table, 0, sizeof table);
        seed = 2026;
        unsigned int n = (unsigned int)(loads[t] * M);
        for (unsigned int i = 0; i < n; i++) {
            unsigned int k = rnd();          /* xorshift는 한 바퀴 안에서 같은 값과 0을 내지 않는다 */
            unsigned int j = h(k);
            while (table[j] != EMPTY && table[j] != k) j = (j + 1) & (M - 1);
            table[j] = k;
            keys[i] = k;
        }
        for (int i = 0; i < 1000000; i++) miss[i] = rnd();   /* 이어지는 값이라 표에 없는 키다 */
        unsigned long long probes = 0;
        double t0 = now_ms();
        for (unsigned int i = 0; i < n; i++) {
            unsigned int k = keys[i], j = h(k);
            probes++;
            while (table[j] != k) { j = (j + 1) & (M - 1); probes++; }
        }
        double t1 = now_ms();
        unsigned long long mprobes = 0;
        double t2 = now_ms();
        for (int i = 0; i < 1000000; i++) {
            unsigned int k = miss[i], j = h(k);
            mprobes++;
            while (table[j] != EMPTY) { j = (j + 1) & (M - 1); mprobes++; }
        }
        double t3 = now_ms();
        printf("load %.2f hit %.2f probes %.1f ns | miss %.2f probes %.1f ns\n", loads[t],
               (double)probes / n, (t1 - t0) * 1e6 / n, (double)mprobes / 1000000, (t3 - t2) * 1e6 / 1000000);
    }
    return 0;
}

    선형 탐사의 평균 탐사 수는 이론상 있는 키가 ½(1 + 1/(1−α)), 없는 키가 ½(1 + 1/(1−α)²)이다. 적재율 0.95까지는 잰 값과 거의 같고, 0.99처럼 거의 찬 표에서는 이론보다 컸다(없는 키 이론 5,000.5칸, 잰 값 8,328칸).

    정리¶

    • 해시 함수가 키를 정수로 바꾸고 % m이 칸 번호를 정한다. 같은 키는 같은 칸으로 가므로 넣기와 찾기가 평균 O(1)이다.
    • 충돌은 피할 수 없다. 체이닝은 칸마다 목록을 달고, 선형 탐사는 다음 빈칸을 쓴다. 선형 탐사에서 지울 때는 "지운 자리" 표시를 남긴다.
    • 적재율이 높으면 느려진다. 정한 적재율을 넘으면 칸을 늘리고 모든 키를 다시 넣는다.
